"""Run the Bayesian Impossible Bottle pipeline and verify it."""
import json, numpy as np
from bottle_pipeline import run
from bottle_controller import BottleController

ctl = BottleController()
print("P(shipType | bottle=demijohn):", {k[:22]: round(v, 3) for k, v in ctl.query("shipType", {"bottleShape": "demijohn with wide belly"}).items()})
print("P(light | table=dark oak):", {k[:30]: round(v, 3) for k, v in ctl.query("light", {"table": "dark oak table, grain left-to-right, worn finish"}).items()})
print("learned structure:", ctl.bn.parents)

res, img, layers = run(evidence={"bottleShape": "classic round, long neck"}, seed=1, outdir="bottle_out", epochs=40)
print("\nproposed config:", json.dumps(res["config"], indent=1))
print("scene log-prob:", round(res["scene_logprob"], 3))
print("per-layer mean uncertainty:", {k: round(v, 4) for k, v in res["layer_uncertainty"].items()})
print("checks:", res["checks"]); print(res["psd"])
ok = all(res["checks"].values()); print("ALL CHECKS PASS" if ok else "CHECK FAILURE")
